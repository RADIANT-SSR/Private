% Shared LaTeX preamble for every volume of the RADIANT manual suite.
%
% Included by scripts/build_manual.py as the SECOND --include-in-header file; the
% first is a generated snippet defining \radiantrunninghead (the volume title).
% The \providecommand below keeps this file usable on its own.
%
% Design constraints:
%   * One template, four volumes — no per-volume forks (Rule 27).
%   * Cover identity stays minimal (ruling Q6): the cover is the report class's
%     own \maketitle page, fed title / subtitle / author / version+date metadata.
%     No logo, no document numbers, no distribution markings are added here.
%   * Optional packages are guarded with \IfFileExists so a BasicTeX-class install
%     degrades to plain formatting instead of failing the build.

\providecommand{\radiantrunninghead}{RADIANT}

\usepackage{etoolbox}

% --- Running heads and feet -------------------------------------------------
% Left: current chapter (\leftmark). Right: volume title. Foot: page number.
\usepackage{fancyhdr}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\sffamily\nouppercase{\leftmark}}
\fancyhead[R]{\small\sffamily\radiantrunninghead}
\fancyfoot[C]{\small\sffamily\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\renewcommand{\footrulewidth}{0pt}
% Chapter-opening and TOC pages use `plain`; give them the same foot, no rule.
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\small\sffamily\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}

% --- Headings in the sans face ----------------------------------------------
% \raggedright plus the infinite hyphenation penalties in the chapter <format>
% group keep display chapter titles from breaking mid-word ("Wavefront-Error
% Bud-get" in Volume IV — CU-370 IV-026); a title too long for one line wraps at
% a space instead.
\IfFileExists{titlesec.sty}{%
  \usepackage{titlesec}%
  \titleformat{\chapter}[display]%
    {\sffamily\bfseries\raggedright\hyphenpenalty=10000\exhyphenpenalty=10000}%
    {\normalsize\sffamily\chaptertitlename\ \thechapter}{8pt}{\Huge}%
  \titlespacing*{\chapter}{0pt}{0pt}{24pt}%
  \titleformat*{\section}{\sffamily\Large\bfseries}%
  \titleformat*{\subsection}{\sffamily\large\bfseries}%
  \titleformat*{\subsubsection}{\sffamily\normalsize\bfseries}%
}{}

% --- Running head on unnumbered chapters ------------------------------------
% Front- and back-matter chapters are staged {.unnumbered}, so pandoc emits
% \chapter*, which — unlike \chapter — sets no header mark. The running head then
% kept whatever the previous page left behind: "Contents" across the whole
% Notation chapter of Volume I (CU-370 I-009). Patch the class's starred-chapter
% hook to set the mark the way \@chapter does for numbered chapters. Placed after
% the titlesec block above, so it wraps whatever definition titlesec leaves.
\makeatletter
\let\radiant@orig@schapter\@schapter
\def\@schapter#1{\radiant@orig@schapter{#1}\markboth{#1}{#1}}
\makeatother

% --- Table of contents ------------------------------------------------------
% The report class allots 2.3em to a section number in the TOC, which a two-digit
% chapter overruns: "12.10Schema Introspection" ran together in Volume III
% (CU-370 III-020). Widen the number box (and the subsection one, for symmetry if
% toc-depth is ever raised).
\makeatletter
\renewcommand*\l@section{\@dottedtocline{1}{1.5em}{3.4em}}
\renewcommand*\l@subsection{\@dottedtocline{2}{4.9em}{4.2em}}
\makeatother

% --- Tables -----------------------------------------------------------------
% The theory chapters and the bound architecture specs carry wide GFM tables
% (8+ columns of prose). Pandoc emits them as longtable with \real{} widths;
% shrinking the body font and the intercolumn padding is what keeps them inside
% the text block instead of overfull.
\AtBeginEnvironment{longtable}{\footnotesize\setlength{\tabcolsep}{4pt}}
\AtBeginEnvironment{tabular}{\footnotesize\setlength{\tabcolsep}{4pt}}
\setlength{\LTleft}{0pt}
\setlength{\LTright}{0pt}
\renewcommand{\arraystretch}{1.15}

% --- Code listings ----------------------------------------------------------
% fvextra adds line breaking to the Verbatim environments pandoc generates for
% fenced code. Redefined at \begin{document} so these win over pandoc's own
% definitions, which are emitted later in the preamble than this include.
\IfFileExists{fvextra.sty}{\usepackage{fvextra}}{}
\AtBeginDocument{%
  \IfFileExists{fvextra.sty}{%
    \DefineVerbatimEnvironment{Highlighting}{Verbatim}%
      {commandchars=\\\{\},breaklines,breakanywhere,breaksymbolleft={},fontsize=\small}%
    \DefineVerbatimEnvironment{verbatim}{Verbatim}%
      {breaklines,breakanywhere,fontsize=\small}%
  }{}%
  % Pandoc's Shaded environment paints this colour behind highlighted blocks.
  \ifdefined\definecolor\definecolor{shadecolor}{RGB}{247,247,247}\fi
  % Numbered PDF bookmarks matching the numbered sections.
  \ifdefined\hypersetup\hypersetup{bookmarksnumbered=true,bookmarksopen=true}\fi
}

% --- Figures ----------------------------------------------------------------
% +implicit_figures (manual.yaml) turns a lone image paragraph into a LaTeX
% figure so the Markdown alt text prints as a numbered caption. A figure is a
% float, and the manual's prose says "the figure below" — pin the floats where
% they are written so the numbering adds captions without reordering anything.
\IfFileExists{float.sty}{%
  \usepackage{float}%
  \floatplacement{figure}{H}%
}{}

% --- Typographic polish -----------------------------------------------------
\widowpenalty=10000
\clubpenalty=10000
\raggedbottom

% --- Unicode coverage for grandfathered source characters --------------------
% The bound chapters carry Unicode super/subscripts and math symbols in prose
% (grandfathered per the plan — no churn rewrites). TeX Gyre Termes lacks the
% glyphs, and XeTeX drops a missing character SILENTLY, so "e⁻" would typeset
% as "e" (first real PDF build, 2026-09-17: 327 dropped glyphs across three
% volumes). Map each to its LaTeX equivalent instead; the builder now also
% fails the build if xelatex reports any character this list does not cover.
%
% A *wrong*-glyph substitution is worse still, because xelatex reports nothing at
% all: U+00B0 DEGREE SIGN reaches the math font (mathspec routes math characters
% to Latin Modern, which has no degree sign) and byte 0xB0 of that font's T1
% encoding is "ř". Every `$\eta = 30°$` in Volumes I, III and IV printed "30ř"
% (CU-370 X-01). The mapping below routes ° to \circ in math and to the text
% font's own degree sign in prose, where it already rendered correctly.
% scripts/build_manual.py::unmapped_risk_chars fails the build if it is removed.
\usepackage{amssymb}
\IfFileExists{newunicodechar.sty}{%
  \usepackage{newunicodechar}%
  \newunicodechar{°}{\ifmmode^{\circ}\else\textdegree\fi}%
  \newunicodechar{⁻}{\textsuperscript{--}}%
  \newunicodechar{⁰}{\textsuperscript{0}}%
  \newunicodechar{⁴}{\textsuperscript{4}}%
  \newunicodechar{⁵}{\textsuperscript{5}}%
  \newunicodechar{⁶}{\textsuperscript{6}}%
  \newunicodechar{⁷}{\textsuperscript{7}}%
  \newunicodechar{⁸}{\textsuperscript{8}}%
  \newunicodechar{⁹}{\textsuperscript{9}}%
  \newunicodechar{₀}{\textsubscript{0}}%
  \newunicodechar{₁}{\textsubscript{1}}%
  \newunicodechar{₂}{\textsubscript{2}}%
  \newunicodechar{₃}{\textsubscript{3}}%
  \newunicodechar{₄}{\textsubscript{4}}%
  \newunicodechar{↔}{\ensuremath{\leftrightarrow}}%
  \newunicodechar{⇒}{\ensuremath{\Rightarrow}}%
  \newunicodechar{∫}{\ensuremath{\int}}%
  \newunicodechar{∏}{\ensuremath{\prod}}%
  \newunicodechar{≫}{\ensuremath{\gg}}%
  \newunicodechar{≪}{\ensuremath{\ll}}%
  \newunicodechar{≡}{\ensuremath{\equiv}}%
  \newunicodechar{≳}{\ensuremath{\gtrsim}}%
  \newunicodechar{≲}{\ensuremath{\lesssim}}%
  \newunicodechar{∝}{\ensuremath{\propto}}%
  \newunicodechar{∈}{\ensuremath{\in}}%
  \newunicodechar{⊥}{\ensuremath{\perp}}%
  \newunicodechar{′}{\ensuremath{{}^{\prime}}}%
  \newunicodechar{″}{\ensuremath{{}^{\prime\prime}}}%
}{}

% --- GUI-chrome symbols ------------------------------------------------------
% The User's Guide quotes interface chrome verbatim (menu arrows, the Inspector
% and Manage buttons, badge marks). Termes has none of these glyphs; DejaVu Sans
% (TeX-Live/MiKTeX-bundled, like the mono face) has them all, so map each to a
% one-off DejaVu run rather than to a lookalike math symbol.
\newfontfamily{\radiantsymbol}{DejaVuSans}[Extension=.ttf, Scale=MatchLowercase]
\IfFileExists{newunicodechar.sty}{%
  \newunicodechar{▸}{{\radiantsymbol ▸}}%
  \newunicodechar{◈}{{\radiantsymbol ◈}}%
  \newunicodechar{⚙}{{\radiantsymbol ⚙}}%
  \newunicodechar{⚠}{{\radiantsymbol ⚠}}%
  \newunicodechar{⚡}{{\radiantsymbol ⚡}}%
  \newunicodechar{✓}{{\radiantsymbol ✓}}%
  \newunicodechar{✕}{{\radiantsymbol ✕}}%
}{}
